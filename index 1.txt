    <script>


        let isEditing = false;


        const editableElements =
            document.querySelectorAll(".editable");



        /* =========================
           EDIT MODE
        ========================== */


        function toggleEditMode() {

            isEditing = !isEditing;


            editableElements.forEach(
                function(element) {

                    element.contentEditable =
                        isEditing;

                }
            );


            document.body.classList.toggle(
                "editing",
                isEditing
            );


            const editButton =
                document.querySelector(
                    ".edit-toggle-btn"
                );


            const saveButton =
                document.querySelector(
                    ".save-btn"
                );


            const resetButton =
                document.querySelector(
                    ".reset-btn"
                );


            if (isEditing) {


                editButton.innerHTML =
                    "✏️ Edit Mode: ON";


                editButton.style.background =
                    "#2e7d32";


                saveButton.style.display =
                    "block";


                resetButton.style.display =
                    "block";


            } else {


                editButton.innerHTML =
                    "✏️ Edit Text Mode: OFF";


                editButton.style.background =
                    "#111";


                saveButton.style.display =
                    "none";


                resetButton.style.display =
                    "none";

            }

        }



        /* =========================
           SAVE CHANGES
        ========================== */


        function saveChanges() {


            const savedData = [];


            editableElements.forEach(
                function(element) {

                    savedData.push(
                        element.innerHTML
                    );

                }
            );


            localStorage.setItem(

                "sonaliTailorWebsite",

                JSON.stringify(savedData)

            );


            alert(
                "✅ Changes Saved Successfully!"
            );

        }



        /* =========================
           LOAD SAVED CHANGES
        ========================== */


        function loadChanges() {


            const savedData =
                localStorage.getItem(
                    "sonaliTailorWebsite"
                );


            if (!savedData) {

                return;

            }


            try {


                const data =
                    JSON.parse(savedData);


                editableElements.forEach(

                    function(element, index) {


                        if (
                            data[index] !==
                            undefined
                        ) {

                            element.innerHTML =
                                data[index];

                        }

                    }

                );


            } catch (error) {


                console.log(
                    "Saved data could not be loaded."
                );

            }

        }



        /* =========================
           RESET CHANGES
        ========================== */


        function resetChanges() {


            const confirmReset = confirm(

                "আপনি কি সব saved changes Reset করতে চান?"

            );


            if (confirmReset) {


                localStorage.removeItem(

                    "sonaliTailorWebsite"

                );


                location.reload();

            }

        }



        /* =========================
           LOAD DATA WHEN PAGE OPENS
        ========================== */


        loadChanges();


    </script>
